   176→   176→                {lang === "bn" ? "সুযোগটি লোড হচ্ছে…" : "Loading opportunity…"}
   177→   177→              </DialogTitle>
   178→   178→              <div className="mx-auto h-12 w-12 animate-pulse rounded-full bg-nx-navy-100" />
   179→   179→              <p className="text-sm text-slate-500">
   180→   180→                {lang === "bn" ? "সুযোগটি লোড হচ্ছে…" : "Loading opportunity…"}
