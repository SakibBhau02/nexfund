    30→/**
    31→ * Insight article reader dialog (R4-3) — the "Read the guide" cards open the
    32→ * full bilingual article here instead of teasing the investor registration.
    33→ *
    34→ * Architecture note (R6): all per-article state (share, reading progress,
    35→ * mini-TOC position, minutes-left) lives in <ReaderContent>, which is keyed
    36→ * by slug and mounted inside DialogContent — Radix unmounts the content when
    37→ * the dialog closes, so every open starts from a clean slate without any
    38→ * reset effects.
    39→ */
    40→export function InsightDialog() {
    41→  const { t } = useLanguage();
    42→  const dialog = useDialogStore((s) => s.dialog);
    43→  const slug = useDialogStore((s) => s.insightSlug);
    44→  const close = useDialogStore((s) => s.close);
    45→  const openInsight = useDialogStore((s) => s.openInsight);
    46→  const isOpen = dialog === "insight";
    47→
    48→  /* the scroll host is DialogContent itself — shared with ReaderContent */
    49→  const scrollElRef = useRef<HTMLDivElement | null>(null);
    50→
    51→  /* R5: deep-link — #insight=<slug> opens the article directly (permalink) */
    52→  useEffect(() => {
    53→    const m = /^#insight=([a-z0-9-]+)$/i.exec(window.location.hash);
    54→    if (!m || !ARTICLES[m[1]]) return;
    55→    // apply post-hydration via rAF so hydration markup stays consistent (simulator pattern)
    56→    const raf = requestAnimationFrame(() => openInsight(m[1]));
    57→    return () => cancelAnimationFrame(raf);
    58→  }, [openInsight]);
    59→
    60→  const card = INSIGHTS.articles.find((a) => a.slug === slug);
    61→  const article = slug ? ARTICLES[slug] : undefined;
    62→
    63→  /* Unknown slug → simple not-found state */
    64→  if (isOpen && !article) {
    65→    return (
    66→      <Dialog open={isOpen} onOpenChange={(v) => !v && close()}>
    67→        <DialogContent
    68→          className="rounded-3xl p-0 sm:max-w-[440px]"
    69→          aria-describedby={undefined}
    70→        >
    71→          <div className="p-8 text-center">
    72→            <FileQuestion className="mx-auto h-10 w-10 text-nx-cyan-500" aria-hidden="true" />
    73→            <DialogTitle className="mt-3 text-base font-bold leading-relaxed text-nx-navy-900">
    74→              {t(READER.notFound)}
    75→            </DialogTitle>
    76→          </div>
    77→        </DialogContent>
    78→      </Dialog>
    79→    );
    80→  }
    81→
    82→  if (!card || !article || !slug) return null;
    83→
    84→  return (
    85→    <Dialog open={isOpen} onOpenChange={(v) => !v && close()}>
    86→      <DialogContent
    87→        ref={scrollElRef}
    88→        aria-describedby={undefined}
    89→        className="nx-scroll max-h-[90vh] gap-0 overflow-y-auto rounded-3xl p-0 sm:max-w-[720px] [&_[data-slot=dialog-close]]:rounded-full [&_[data-slot=dialog-close]]:bg-white/15 [&_[data-slot=dialog-close]]:p-1.5 [&_[data-slot=dialog-close]]:text-white [&_[data-slot=dialog-close]]:backdrop-blur [&_[data-slot=dialog-close]]:transition-colors [&_[data-slot=dialog-close]]:hover:bg-white/30"
    90→      >
    91→        {isOpen ? (
    92→          <ReaderContent key={slug} slug={slug} card={card} article={article} scrollElRef={scrollElRef} />
    93→        ) : null}
    94→      </DialogContent>
    95→    </Dialog>
    96→  );
    97→}

    99→/** Per-article reader body — keyed by slug, so state resets on remount. */
   100→function ReaderContent({
   101→  slug,
   102→  card,
   103→  article,
   104→  scrollElRef,
   105→}: {
   106→  slug: string;
   107→  card: ReaderCard;
   108→  article: ReaderArticle;
   109→  scrollElRef: React.RefObject<HTMLDivElement | null>;
   110→}) {
   111→  const { lang, t } = useLanguage();
   112→  const open = useDialogStore((s) => s.open);
   113→  const openInvestor = useDialogStore((s) => s.openInvestor);
   114→  const reduce = useReducedMotion();
   115→
   116→  /* ── R5: share — native share sheet where available, clipboard fallback ── */
   117→  const [shareState, setShareState] = useState<"idle" | "copied" | "failed">("idle");
   118→  const shareTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   119→
   120→  /* ── R8: font-size control (A−/A+) — 3 steps, persisted so the choice
   121→     survives reopen. State lives in the keyed ReaderContent, which keeps it
   122→     across BN⇄EN (dialogs sit outside the language cross-fade). */
   123→  const [fontStep, setFontStep] = useState<"-1" | "0" | "1">(() => {
   124→    try {
   125→      const v = window.localStorage.getItem("nx-reader-font");
   126→      return v === "-1" || v === "1" ? v : "0";
   127→    } catch {
   128→      return "0";
   129→    }
   130→  });
   131→  const changeFont = (dir: -1 | 1) => {
   132→    const next = Math.max(-1, Math.min(1, Number(fontStep) + dir));
   133→    const v = (next === 0 ? "0" : next === -1 ? "-1" : "1") as "-1" | "0" | "1";
   134→    setFontStep(v);
   135→    try {
   136→      window.localStorage.setItem("nx-reader-font", v);
   137→    } catch {
   138→      /* storage unavailable — the step still applies for this session */
   139→    }
   140→  };
   141→
   142→  useEffect(
   143→    () => () => {
   144→      if (shareTimer.current) clearTimeout(shareTimer.current);
   145→    },
   146→    []
   147→  );
   148→
   149→  const share = async () => {
   150→    const url = `${window.location.origin}${window.location.pathname}#insight=${slug}`;
   151→    if (typeof navigator.share === "function") {
   152→      try {
   153→        await navigator.share({ title: document.title, url });
   154→        return; // share sheet handled it — no local state change needed
   155→      } catch {
   156→        // user dismissed the sheet, or share failed → fall through to clipboard
   157→      }
   158→    }
   159→    let ok = false;
   160→    try {
   161→      await navigator.clipboard.writeText(url);
   162→      ok = true;
   163→    } catch {
   164→      try {
   165→        const ta = document.createElement("textarea");
   166→        ta.value = url;
   167→        ta.style.position = "fixed";
   168→        ta.style.opacity = "0";
   169→        document.body.appendChild(ta);
   170→        ta.focus();
   171→        ta.select();
   172→        ok = document.execCommand("copy");
   173→        document.body.removeChild(ta);
   174→      } catch {
   175→        ok = false;
   176→      }
   177→    }
   178→    setShareState(ok ? "copied" : "failed");
   179→    if (shareTimer.current) clearTimeout(shareTimer.current);
   180→    shareTimer.current = setTimeout(() => setShareState("idle"), 2600);
   181→  };
   182→
   183→  /* ── Reading progress + R6 mini-TOC tracking + minutes-left ── */
   184→  const progressWrapRef = useRef<HTMLDivElement>(null);
   185→  const progressBarRef = useRef<HTMLDivElement>(null);
   186→  const secRefs = useRef<(HTMLElement | null)[]>([]);
   187→  const [activeSec, setActiveSec] = useState(0);
   188→  const [minLeft, setMinLeft] = useState<number | null>(null);
   189→  const lastLeftRef = useRef(-1); // last displayed minutes-left
   190→
   191→  // native scroll listener on the dialog's scroll host (DialogContent) —
   192→  // setState only fires from the event callback, never from the effect body
   193→  useEffect(() => {
   194→    const host = scrollElRef.current;
   195→    if (!host) return;
   196→    const onScroll = () => {
   197→      const denom = host.scrollHeight - host.clientHeight;
   198→      const p = denom > 0 ? Math.min(1, Math.max(0, host.scrollTop / denom)) : 0;
   199→      if (progressBarRef.current) progressBarRef.current.style.transform = `scaleX(${p})`;
   200→      if (progressWrapRef.current) {
   201→        progressWrapRef.current.setAttribute("aria-valuenow", String(Math.round(p * 100)));
   202→      }
   203→
   204→      /* track the section currently at the top of the viewport */
   205→      const secs = secRefs.current;
   206→      if (secs.length > 0) {
   207→        const hostTop = host.getBoundingClientRect().top;
   208→        let idx = 0;
   209→        for (let i = 0; i < secs.length; i++) {
   210→          const s = secs[i];
   211→          if (s && s.getBoundingClientRect().top - hostTop <= 160) idx = i;
   212→        }
   213→        const found = idx;
   214→        setActiveSec((prev) => (prev === found ? prev : found));
   215→      }
   216→
   217→      /* remaining reading time — re-renders only when the minute changes */
   218→      const left = Math.max(0, Math.ceil(card.minutes * (1 - p)));
   219→      if (left !== lastLeftRef.current) {
   220→        lastLeftRef.current = left;
   221→        setMinLeft(left);
   222→      }
   223→    };
   224→    host.addEventListener("scroll", onScroll, { passive: true });
   225→    return () => host.removeEventListener("scroll", onScroll);
   226→  }, [scrollElRef, card.minutes]);
   227→
   228→  /* R6: TOC jump — rect math (robust vs offsetParent), no page-side scroll */
   229→  const jumpTo = (i: number) => {
   230→    const el = scrollElRef.current;
   231→    const s = secRefs.current[i];
   232→    if (!el || !s) return;
   233→    const delta = s.getBoundingClientRect().top - el.getBoundingClientRect().top;
   234→    el.scrollTo({ top: el.scrollTop + delta - 14, behavior: reduce ? "auto" : "smooth" });
   235→    setActiveSec(i);
   236→  };
   237→
   238→  const updated = lang === "bn" ? bnNum(article.updated) : article.updated;
   239→
   240→  /* R9: print sheet date — native locale, mirroring the compare print sheet */
   241→  const preparedDate = new Date().toLocaleDateString(lang === "bn" ? "bn-BD" : "en-GB", {
   242→    day: "numeric",
   243→    month: "long",
   244→    year: "numeric",
   245→  });
   246→
   247→  return (
   248→    <>
   249→      {/* ── Reading progress (thin, sticky at very top) + remaining-time pill ── */}
   250→      <div
   251→        ref={progressWrapRef}
   252→        role="progressbar"
   253→        aria-label={t(READER.scrollProgressAria)}
   254→        aria-valuemin={0}
   255→        aria-valuemax={100}
   256→        aria-valuenow={0}
   257→        className="sticky top-0 z-20 h-[3px] w-full"
   258→      >
   259→        <div
   260→          ref={progressBarRef}
   261→          aria-hidden="true"
   262→          className="nx-progress-gradient h-full w-full"
   263→          style={{ transform: "scaleX(0)" }}
   264→        />
   265→        {/* R6: floating minutes-left pill — rides with the sticky bar */}
   266→        <span
   267→          aria-hidden="true"
   268→          className="nx-num pointer-events-none absolute right-4 top-[9px] inline-flex items-center gap-1 rounded-full border border-nx-navy-100 bg-white/92 px-2.5 py-1 text-[10px] font-extrabold text-nx-navy-700 shadow-[0_6px_16px_-8px_rgba(6,31,74,0.35)] backdrop-blur"
   269→        >
   270→          <Clock className="h-3 w-3 text-nx-cyan-500" />
   271→          {t(READERTOC.remaining(minLeft ?? card.minutes))}
   272→        </span>
   273→      </div>
   274→
   275→      <motion.div
   276→        initial={{ opacity: 0, y: 12 }}
   277→        animate={{ opacity: 1, y: 0 }}
   278→        transition={{ duration: 0.32, ease: "easeOut" }}
   279→      >
   280→        {/* ── Hero image strip ── */}
   281→        <div className="relative h-44 w-full overflow-hidden">
   282→          <Image
   283→            src={card.image}
   284→            alt={t(card.title)}
   285→            fill
   286→            sizes="(max-width: 768px) 100vw, 720px"
   287→            className="object-cover"
   288→          />
   289→          <div
   290→            className="absolute inset-0 bg-gradient-to-t from-nx-navy-900/85 via-nx-navy-900/35 to-transparent"
   291→            aria-hidden="true"
   292→          />
   293→          <div className="absolute inset-x-0 bottom-0 flex flex-wrap items-center gap-2 p-5">
   294→            <span className="rounded-full bg-white/90 px-3 py-1 text-[11px] font-bold text-nx-navy-800 backdrop-blur">
   295→              {t(card.category)}
   296→            </span>
   297→            <span className="inline-flex items-center gap-1.5 rounded-full border border-white/25 bg-white/15 px-3 py-1 text-[11px] font-bold text-white backdrop-blur">
   298→              <Clock className="h-3 w-3" aria-hidden="true" />
   299→              {t(INSIGHTS.readTime(card.minutes))}
   300→            </span>
   301→            <span className="nx-num inline-flex items-center rounded-full border border-white/25 bg-white/15 px-3 py-1 text-[11px] font-bold text-white backdrop-blur">
   302→              {t(READER.updatedLabel)} {updated}
   303→            </span>
   304→            {/* R5: share permalink — native sheet on mobile, clipboard elsewhere */}
   305→            <button
   306→              type="button"
   307→              onClick={share}
   308→              aria-live="polite"
   309→              className="ml-auto inline-flex items-center gap-1.5 rounded-full border border-white/25 bg-white/15 px-3 py-1 text-[11px] font-bold text-white backdrop-blur transition-colors hover:bg-white/30 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-white"
   310→            >
   311→              {shareState === "copied" ? (
   312→                <Check className="h-3 w-3" aria-hidden="true" />
   313→              ) : (
   314→                <Share2 className="h-3 w-3" aria-hidden="true" />
   315→              )}
   316→              {shareState === "copied"
   317→                ? t(SHARE.linkCopied)
   318→                : shareState === "failed"
   319→                  ? t(SHARE.copyFailed)
   320→                  : t(SHARE.shareArticle)}
   321→            </button>
   322→            {/* R9: print / save-PDF of the full article (sheet at the end of
   323→                the dialog; the :has()-guarded print CSS takes over) */}
   324→            <button
   325→              type="button"
   326→              onClick={() => window.print()}
   327→              title={t(READER_PRINT.button)}
   328→              className="inline-flex items-center gap-1.5 rounded-full border border-white/25 bg-white/15 px-3 py-1 text-[11px] font-bold text-white backdrop-blur transition-colors hover:bg-white/30 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-white"
   329→            >
   330→              <Printer className="h-3 w-3" aria-hidden="true" />
   331→              {t(READER_PRINT.button)}
   332→            </button>
   333→          </div>
   334→        </div>
   335→
   336→        {/* ── Title + short answer ── */}
   337→        <DialogHeader className="px-6 pb-2 pt-5 sm:px-8">
   338→          <DialogTitle className="text-xl font-extrabold leading-snug text-nx-navy-900 sm:text-2xl">
   339→            {t(card.title)}
   340→          </DialogTitle>
   341→          <div className="mt-3 rounded-xl bg-nx-cyan-50 p-4 text-left">
   342→            <p className="text-[10px] font-extrabold tracking-[0.14em] text-nx-cyan-700 uppercase">
   343→              {t(INSIGHTS.shortAnswer)}
   344→            </p>
   345→            <p className="mt-1 text-[13px] leading-relaxed text-nx-ink/80">{t(card.short)}</p>
   346→          </div>
   347→        </DialogHeader>
   348→
   349→        {/* ── R6: sticky mini-TOC — numbered jump chips per section ──
   350→            R8: row restructured — scrollable chips (flex-1) + pinned A−/A+
   351→            font-size control on the right, always reachable on mobile. */}
   352→        <nav
   353→          aria-label={t(READERTOC.tocLabel)}
   354→          className="sticky top-[3px] z-10 border-b border-nx-navy-100 bg-white/92 px-6 py-2.5 backdrop-blur sm:px-8"
   355→        >
   356→          <div className="flex items-center gap-2">
   357→            <div className="nx-scroll flex min-w-0 flex-1 items-center gap-1.5 overflow-x-auto pb-0.5">
   358→              <span className="shrink-0 pr-1 text-[10px] font-extrabold tracking-[0.14em] text-slate-400 uppercase">
   359→                {t(READERTOC.tocLabel)}
   360→              </span>
   361→              {article.sections.map((sec, i) => {
   362→                const num = String(i + 1).padStart(2, "0");
   363→                const active = activeSec === i;
   364→                return (
   365→                  <button
   366→                    key={sec.h.en}
   367→                    type="button"
   368→                    aria-current={active ? "true" : undefined}
   369→                    title={t(sec.h)}
   370→                    onClick={() => jumpTo(i)}
   371→                    className={cn(
   372→                      "nx-num shrink-0 rounded-full border px-3 py-1 text-[11px] font-extrabold transition-all duration-200 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-nx-cyan-400",
   373→                      active
   374→                        ? "border-nx-navy-700 bg-nx-navy-700 text-white shadow-[0_8px_18px_-8px_rgba(10,58,143,0.7)]"
   375→                        : "border-nx-navy-200 bg-white text-nx-navy-600 hover:border-nx-cyan-400 hover:text-nx-cyan-700"
   376→                    )}
   377→                  >
   378→                    <span className="sr-only">{t(sec.h)}</span>
   379→                    {lang === "bn" ? bnNum(num) : num}
   380→                  </button>
   381→                );
   382→              })}
   383→            </div>
   384→            {/* R8: reading-comfort font-size control */}
   385→            <div
   386→              role="group"
   387→              aria-label={t(READER_FONT.label)}
   388→              title={t(READER_FONT.reset)}
   389→              className="flex shrink-0 items-center gap-1"
   390→            >
   391→              <button
   392→                type="button"
   393→                onClick={() => changeFont(-1)}
   394→                disabled={fontStep === "-1"}
   395→                aria-label={t(READER_FONT.smaller)}
   396→                title={t(READER_FONT.smaller)}
   397→                className="flex h-7 w-7 items-center justify-center rounded-full border border-nx-navy-200 bg-white text-nx-navy-600 transition-colors hover:border-nx-cyan-400 hover:text-nx-cyan-700 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-nx-cyan-400 disabled:cursor-not-allowed disabled:opacity-40 disabled:hover:border-nx-navy-200 disabled:hover:text-nx-navy-600"
   398→              >
   399→                <AArrowDown className="h-3.5 w-3.5" aria-hidden="true" />
   400→              </button>
   401→              <button
   402→                type="button"
   403→                onClick={() => changeFont(1)}
   404→                disabled={fontStep === "1"}
   405→                aria-label={t(READER_FONT.larger)}
   406→                title={t(READER_FONT.larger)}
   407→                className="flex h-7 w-7 items-center justify-center rounded-full border border-nx-navy-200 bg-white text-nx-navy-600 transition-colors hover:border-nx-cyan-400 hover:text-nx-cyan-700 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-nx-cyan-400 disabled:cursor-not-allowed disabled:opacity-40 disabled:hover:border-nx-navy-200 disabled:hover:text-nx-navy-600"
   408→              >
   409→                <AArrowUp className="h-3.5 w-3.5" aria-hidden="true" />
   410→              </button>
   411→            </div>
   412→          </div>
   413→        </nav>
   414→
   415→        {/* ── Article body (R8: scales with the A−/A+ control) ── */}
   416→        <div
   417→          className={cn(
   418→            "px-6 pb-6 pt-4 sm:px-8",
   419→            fontStep === "-1" && "nx-reader-sm",
   420→            fontStep === "1" && "nx-reader-lg"
   421→          )}
   422→        >
   423→          {article.sections.map((sec, i) => {
   424→            const num = String(i + 1).padStart(2, "0");
   425→            return (
   426→              <section
   427→                key={sec.h.en}
   428→                ref={(el) => {
   429→                  secRefs.current[i] = el;
   430→                }}
   431→                className={i > 0 ? "mt-7" : undefined}
   432→              >
   433→                <div className="flex items-baseline gap-3">
   434→                  <span className="nx-num text-sm font-extrabold text-nx-cyan-500" aria-hidden="true">
   435→                    {lang === "bn" ? bnNum(num) : num}
   436→                  </span>
   437→                  <h3 className="text-base font-extrabold leading-snug text-nx-navy-900 sm:text-lg">
   438→                    {t(sec.h)}
   439→                  </h3>
   440→                </div>
   441→                {t(sec.body)
   442→                  .split("\n\n")
   443→                  .map((p, j) => (
   444→                    <p key={j} className="mt-2.5 text-sm leading-relaxed text-slate-600">
   445→                      {p}
   446→                    </p>
   447→                  ))}
   448→                {sec.list && (
   449→                  <ul className="mt-3 space-y-2.5">
   450→                    {sec.list.map((li) => (
   451→                      <li
   452→                        key={li.en}
   453→                        className="flex items-start gap-2.5 text-sm leading-relaxed text-slate-600"
   454→                      >
   455→                        <span
   456→                          className="mt-[7px] h-1.5 w-1.5 shrink-0 rounded-full bg-nx-cyan-500"
   457→                          aria-hidden="true"
   458→                        />
   459→                        {t(li)}
   460→                      </li>
   461→                    ))}
   462→                  </ul>
   463→                )}
   464→              </section>
   465→            );
   466→          })}
   467→
   468→          {/* ── Glossary chips ── */}
   469→          <div className="mt-8 flex flex-wrap items-center gap-2 border-t border-nx-navy-100 pt-5">
   470→            <span className="flex items-center gap-1 text-[11px] font-bold tracking-wide text-slate-400 uppercase">
   471→              <BookMarked className="h-3 w-3" aria-hidden="true" />
   472→              {t(READER.termsLabel)}
   473→            </span>
   474→            {article.terms.map((term) => (
