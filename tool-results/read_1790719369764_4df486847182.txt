   169→                </DialogTitle>
   170→                <Tooltip>
   171→                  <TooltipTrigger asChild>

   173→                      <Sparkles className="h-3 w-3" aria-hidden="true" />

   176→                  </TooltipTrigger>
   177→                  <TooltipContent className="max-w-[240px] text-xs leading-relaxed">

   295→                          <span className="nx-num font-extrabold text-nx-navy-700">

   338→        {/* ── Advisor review ── */}

   342→            {t(OPP_DLG.advisorTitle)}
   343→          </p>
   344→          <p className="mt-2 text-sm leading-relaxed text-nx-ink/85">
   345→            {lang === "bn" ? o.advisorNoteBn : o.advisorNote}

   355→                initial={{ opacity: 0, y: 10 }}

   511→/* ── R4-1: per-listing illustrative scenarios (blueprint §5.5 —
   512→     "downside scenario chart alongside the upside") ── */
   513→function ScenariosPanel({ o }: { o: OpportunityDTO }) {
   514→  const { t, lang } = useLanguage();
   515→  const close = useDialogStore((s) => s.close);
   516→
   517→  const listing = SCEN.listings[o.slug];
   518→  const mode: "equity" | "revshare" = listing?.mode ?? "equity";
   519→  const years = listing?.years ?? SCEN.equityFallback.years;
   520→  const ticket = listing?.ticket ?? Math.round((o.seekingMin + o.seekingMax) / 2);
   521→
   522→  const fallback = (key: "down" | "base" | "up"): ScenAssumption => {
   523→    const g = SCEN.equityFallback.baseGrowth;
   524→    return {
   525→      growth: key === "down" ? g - 15 : key === "up" ? g + 8 : g,
   526→      multiple:
   527→        key === "down" ? SCEN.equityFallback.down : key === "up" ? SCEN.equityFallback.up : SCEN.equityFallback.base,
   528→      note: SIM.scenarios[key].desc,
   529→    };
   530→  };
   531→  const assumptions: Record<"down" | "base" | "up", ScenAssumption> = listing
   532→    ? { down: listing.down, base: listing.base, up: listing.up }
   533→    : { down: fallback("down"), base: fallback("base"), up: fallback("up") };
   534→
   535→  const computed = (key: "down" | "base" | "up") => {
   536→    const a = assumptions[key];
   537→    const multiple =
   538→      mode === "revshare" ? a.multiple : Math.pow(1 + (a.growth ?? 0) / 100, years) * a.multiple;
   539→    return { multiple, proceeds: ticket * multiple };
   540→  };
   541→  const res = { down: computed("down"), base: computed("base"), up: computed("up") };
   542→  const maxVal = Math.max(res.up.proceeds, res.base.proceeds, res.down.proceeds, ticket);
   543→  const ticketPct = (ticket / maxVal) * 100;
   544→
   545→  const signedPct = (v: number) => {
   546→    const s = v < 0 ? "−" : v > 0 ? "+" : "";
   547→    return lang === "bn" ? `${s}${bnNum(Math.abs(v))}%` : `${s}${Math.abs(v)}%`;
   548→  };
   549→  const fmtMultiple = (m: number) => (lang === "bn" ? `${bnNum(m.toFixed(2))}×` : `${m.toFixed(2)}×`);
   550→
   551→  const meta: { key: "down" | "base" | "up"; icon: typeof TrendingDown; barClass: string; chipClass: string; lead: boolean }[] = [
   552→    { key: "down", icon: TrendingDown, barClass: "nx-warm-to-loss", chipClass: "text-nx-warn", lead: true },
   553→    { key: "base", icon: Minus, barClass: "bg-nx-navy-600", chipClass: "text-nx-navy-600", lead: false },
   554→    { key: "up", icon: TrendingUp, barClass: "bg-nx-cyan-500", chipClass: "text-nx-cyan-600", lead: false },
   555→  ];
   556→
   557→  const assumptionLine = (a: ScenAssumption) =>
   558→    mode === "revshare"
   559→      ? lang === "bn"
   560→        ? `রিটার্ন ${fmtMultiple(a.multiple)} · ${bnNum(years)} বছর`
   561→        : `return ${fmtMultiple(a.multiple)} · ${years} yrs`
   562→      : lang === "bn"
   563→        ? `বৃদ্ধি ${signedPct(a.growth ?? 0)}/বছর · এক্সিট ${fmtMultiple(a.multiple)} · ${bnNum(years)} বছর`
   564→        : `growth ${signedPct(a.growth ?? 0)}/yr · exit ${fmtMultiple(a.multiple)} · ${years} yrs`;
   565→
   566→  const goToSimulator = () => {
   567→    close();
   568→    // wait for the dialog unmount before smooth-scrolling
   569→    setTimeout(() => {
   570→      document.getElementById("simulator")?.scrollIntoView({ behavior: "smooth" });
   571→    }, 90);
   572→  };
   573→
   574→  return (
   575→    <div className="space-y-4">
   576→      {/* chips: model · horizon · modeled ticket */}
   577→      <div className="flex flex-wrap items-center gap-2">
   578→        <span
   579→          className={cn(
   580→            "rounded-full border px-3 py-1 text-[11px] font-bold",
   581→            mode === "revshare"
   582→              ? "border-nx-cyan-200 bg-nx-cyan-50 text-nx-cyan-700"
   583→              : "border-nx-navy-200 bg-nx-navy-50 text-nx-navy-800"
   584→          )}
   585→        >
   586→          {t(mode === "revshare" ? SCEN.modeRevshare : SCEN.modeEquity)}
   587→        </span>
   588→        <span className="rounded-full border border-nx-navy-100 bg-nx-mist px-3 py-1 text-[11px] font-bold text-slate-600">
   589→          {t(SCEN.horizon(years))}
   590→        </span>
   591→        <span className="nx-num rounded-full bg-nx-navy-900 px-3 py-1 text-[11px] font-extrabold text-white">
   592→          {formatTk(ticket, lang)}
   593→          <span className="ml-1.5 font-semibold text-white/60">{t(SCEN.ticketModeled)}</span>
   594→        </span>
   595→      </div>
   596→
   597→      {mode === "revshare" && (
   598→        <div className="flex items-start gap-2 rounded-xl border border-nx-cyan-200 bg-nx-cyan-50 p-3.5 text-xs leading-relaxed text-nx-ink/80">
   599→          <Info className="mt-0.5 h-3.5 w-3.5 shrink-0 text-nx-cyan-600" aria-hidden="true" />
   600→          {t(SCEN.revshareNote)}
   601→        </div>
   602→      )}
   603→
   604→      <div className="space-y-3.5">
   605→        {meta.map(({ key, icon: Icon, barClass, chipClass, lead }) => {
   606→          const r = res[key];
   607→          const isLoss = r.multiple < 1;
   608→          const pctChange = Math.round((r.multiple - 1) * 100);
   609→          const widthPct = Math.max((r.proceeds / maxVal) * 100, 2);
   610→          return (
   611→            <div
   612→              key={key}
   613→              className={cn(
   614→                "rounded-2xl border p-4",
   615→                lead ? "border-nx-warn/50 bg-nx-warn-bg/50" : "border-nx-navy-100 bg-white"
   616→              )}
   617→            >
   618→              <div className="flex flex-wrap items-baseline justify-between gap-x-3 gap-y-1">
   619→                <p className="flex items-center gap-1.5 text-sm font-extrabold text-nx-navy-900">
   620→                  <Icon className={cn("h-4 w-4", chipClass)} aria-hidden="true" />
   621→                  {t(SIM.scenarios[key].name)}
   622→                  {lead && (
   623→                    <span className="ml-1 rounded-full bg-nx-warn/15 px-2 py-0.5 text-[10px] font-bold text-nx-warn">
   624→                      {lang === "bn" ? "আগে দেখুন" : "look here first"}
   625→                    </span>
   626→                  )}
   627→                </p>
   628→                <p className="nx-num text-[11px] font-semibold text-slate-500">{assumptionLine(assumptions[key])}</p>
   629→              </div>
   630→              <p className="mt-0.5 text-xs leading-snug text-slate-500">{t(assumptions[key].note)}</p>
   631→
   632→              {/* bar with break-even marker */}
   633→              <div className="relative mt-3">
   634→                <div className="h-8 overflow-hidden rounded-xl bg-nx-navy-50">
   635→                  <motion.div
   636→                    initial={{ width: 0 }}
   637→                    animate={{ width: `${widthPct}%` }}
   638→                    transition={{ duration: 0.55, ease: [0.2, 0.8, 0.2, 1] }}
   639→                    className={cn("flex h-full items-center justify-end rounded-xl pr-2.5", barClass)}
   640→                  >
   641→                    {widthPct > 24 && (
   642→                      <span className="nx-num text-[11px] font-extrabold text-white">
   643→                        {formatTk(Math.round(r.proceeds), lang)}
   644→                      </span>
   645→                    )}
   646→                  </motion.div>
   647→                </div>
   648→                {ticketPct > 6 && ticketPct < 97 && (
   649→                  <div
   650→                    className="pointer-events-none absolute inset-y-0"
   651→                    style={{ left: `${ticketPct}%` }}
   652→                    aria-hidden="true"
   653→                  >
   654→                    <div className="h-full w-0 border-l-2 border-dashed border-nx-navy-900/45" />
   655→                  </div>
   656→                )}
   657→              </div>
   658→
   659→              <div className="mt-2.5 flex flex-wrap items-baseline justify-between gap-x-4 gap-y-1">
   660→                <p className="text-[11px] font-bold tracking-wide text-slate-400 uppercase">{t(SIM.exitValue)}</p>
   661→                <p className="flex items-baseline gap-2.5">
   662→                  <span className="nx-num text-sm font-extrabold text-nx-navy-900">
   663→                    {formatTk(Math.round(r.proceeds), lang)}
   664→                  </span>
   665→                  <span
   666→                    className={cn(
   667→                      "nx-num rounded-full px-2 py-0.5 text-[11px] font-extrabold",
   668→                      isLoss ? "bg-nx-danger/10 text-nx-danger" : "bg-nx-verified-bg text-nx-verified"
   669→                    )}
   670→                  >
   671→                    {lang === "bn"
   672→                      ? `${fmtMultiple(r.multiple)} · ${t(SIM.changeLabel)} ${signedPct(pctChange)} ${isLoss ? t(SIM.loss) : t(SIM.gain)}`
   673→                      : `${fmtMultiple(r.multiple)} · ${signedPct(pctChange)} ${isLoss ? t(SIM.loss) : t(SIM.gain)} ${t(SIM.changeLabel)}`}
   674→                  </span>
   675→                </p>
   676→              </div>
   677→            </div>
   678→          );
   679→        })}
   680→      </div>
   681→
   682→      {/* legend + honesty footnote */}
   683→      <p className="flex items-center gap-2 text-[11px] text-slate-500">
   684→        <span className="inline-block h-3.5 w-0 border-l-2 border-dashed border-nx-navy-900/45" aria-hidden="true" />
   685→        {t(SCEN.breakEven)}
   686→      </p>
   687→      <p className="flex items-start gap-2 rounded-xl border border-nx-warn/40 bg-nx-warn-bg px-3.5 py-3 text-[11px] leading-relaxed text-nx-warn">
   688→        <AlertTriangle className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden="true" />
   689→        {t(SCEN.footnote)}
   690→      </p>
   691→
   692→      <button
   693→        onClick={goToSimulator}
   694→        className="nx-arrow-btn inline-flex items-center gap-1.5 rounded-full border-[1.5px] border-nx-navy-200 px-4 py-2 text-xs font-bold text-nx-navy-800 transition-all hover:border-nx-cyan-500 hover:text-nx-cyan-700"
   695→      >
   696→        {t(SCEN.tryYourOwn)}
   697→        <span className="nx-arrow">
   698→          <ArrowUpRight className="h-3.5 w-3.5" aria-hidden="true" />
   699→        </span>
   700→      </button>
   701→    </div>
   702→  );
